<script>
  // Apply creator config
  document.getElementById("opening").innerText = CONFIG.openingText;
  document.getElementById("question").innerText = CONFIG.questionText;
  document.getElementById("note").innerText = CONFIG.loveNote;
  document.getElementById("final").innerText = CONFIG.finalText;
  document.getElementById("song").src = CONFIG.songURL;

  const noBtn = document.getElementById("noBtn");
  const yesBtn = document.getElementById("yesBtn");
  const result = document.getElementById("result");
  const song = document.getElementById("song");

  const noTexts = [
    "Try again 😏",
    "Nope 😌",
    "Not happening 💅",
    "Are you sure? 😳",
    "Nice try 😜"
  ];

  noBtn.addEventListener("mouseover", () => {
    const x = Math.random() * 200 - 100;
    const y = Math.random() * 200 - 100;
    noBtn.style.transform = `translate(${x}px, ${y}px)`;
    noBtn.innerText = noTexts[Math.floor(Math.random() * noTexts.length)];
  });

  yesBtn.addEventListener("click", () => {
    document.getElementById("buttons").classList.add("hidden");
    result.classList.remove("hidden");
    song.play();
    launchHearts();
  });

  function launchHearts() {
    const hearts = document.getElementById("hearts");
    setInterval(() => {
      const heart = document.createElement("div");
      heart.classList.add("heart");
      heart.innerText = "❤️";
      heart.style.left = Math.random() * 100 + "vw";
      heart.style.fontSize = Math.random() * 20 + 15 + "px";
      hearts.appendChild(heart);
      setTimeout(() => heart.remove(), 4000);
    }, 300);
  }
</script>
